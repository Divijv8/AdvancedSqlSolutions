SELECT project_id, employee_id
FROM (
  SELECT p.employee_id, p.project_id, DENSE_RANK() OVER (PARTITION BY p.project_id ORDER BY e.experience_years DESC) as experience_rank
  FROM Project p
  LEFT JOIN Employee e
  ON e.employee_id = p.employee_id
) n
WHERE experience_rank = 1 AND project_id IS NOT NULL;
