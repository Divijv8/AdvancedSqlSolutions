SELECT name
FROM SalesPerson
WHERE sales_id NOT IN (
    SELECT o.sales_id
    FROM Orders o
    LEFT JOIN Company c
    ON c.com_id = o.com_id
    WHERE c.name = 'RED'
);

// Order of Left Join matters here because if we do opposite left join then if none is connected to 'RED' subquery will return NULL.. and using NOT IN with NULL can throw 'UNKNOWN' values meaning no output
